# Ano de entrada × gênero, de 2015 em diante: as duas saídas em CSV, comparadas linha a linha.
$ psql ... -At -F, -c "SELECT year_added, genre, MEASURE(count_catalog_titles), MEASURE(count_titles), round(MEASURE(avg_rating)::numeric, 3) FROM catalog_overview WHERE year_added >= 2015 GROUP BY 1, 2 ORDER BY 1, 2" > cube.csv
$ trino --output-format CSV_UNQUOTED --execute "select c.added_year, g.genre_key, count(distinct c.title_key), count(distinct t.title_key), round(avg(t.rating), 3) from refined.ml.fct_catalog c join refined.ml.bridge_title_genre g on g.title_key = c.title_key left join refined.ml.fct_titles t on t.title_key = c.title_key where c.added_year >= 2015 group by 1, 2 order by 1, 2" > trino.csv
$ python3 cmp.py cube.csv trino.csv   # compara como número (7 = 7.0)
