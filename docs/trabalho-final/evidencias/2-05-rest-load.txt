# SQL que o Cube gera para essa consulta (/v1/sql): a medida do fato atravessa a ponte
# por uma subconsulta de chaves (keys), e não por count(*) sobre o join.
$ curl -sG http://localhost:4000/cubejs-api/v1/sql --data-urlencode 'query=...mesma consulta...'
 WITH 
cte_0 AS (  SELECT "catalog".added_year "catalog_overview__year_added", "title_genre".genre_key "catalog_overview__genre", COUNT(DISTINCT "catalog".title_key) "catalog__count_catalog_titles"  
  FROM  refined.ml.fct_catalog  AS "catalog"
  LEFT JOIN  refined.ml.bridge_title_genre  AS "title_genre" ON "catalog".title_key = "title_genre".title_key
  WHERE ("catalog".added_year >= CAST(? AS DOUBLE)) AND ("title_genre".genre_key IN (?, ?, ?)) GROUP BY 1, 2),
cte_1 AS (  SELECT "catalog".added_year "catalog_overview__year_added", "title_genre".genre_key "catalog_overview__genre", COUNT(DISTINCT "titles".title_key) "titles__count_titles"  
  FROM  refined.ml.fct_catalog  AS "catalog"
  LEFT JOIN  refined.ml.bridge_title_genre  AS "title_genre" ON "catalog".title_key = "title_genre".title_key
  LEFT JOIN  refined.ml.fct_titles  AS "titles" ON "catalog".title_key = "titles".title_key
  WHERE ("catalog".added_year >= CAST(? AS DOUBLE)) AND ("title_genre".genre_key IN (?, ?, ?)) GROUP BY 1, 2),
cte_2 AS (  SELECT "keys"."catalog_overview__year_added" "catalog_overview__year_added", "keys"."catalog_overview__genre" "catalog_overview__genre", avg("titles_key_titles".rating) "titles__avg_rating"  
  FROM (SELECT DISTINCT "titles_key_catalog".added_year "catalog_overview__year_added", "titles_key_title_genre".genre_key "catalog_overview__genre", "titles_key_titles".title_key "titles__title_key"  
  FROM  refined.ml.fct_catalog  AS "titles_key_catalog"
  LEFT JOIN  refined.ml.bridge_title_genre  AS "titles_key_title_genre" ON "titles_key_catalog".title_key = "titles_key_title_genre".title_key
  LEFT JOIN  refined.ml.fct_titles  AS "titles_key_titles" ON "titles_key_catalog".title_key = "titles_key_titles".title_key
  WHERE ("titles_key_catalog".added_year >= CAST(? AS DOUBLE)) AND ("titles_key_title_genre".genre_key IN (?, ?, ?))) AS "keys"
  LEFT JOIN  refined.ml.fct_titles  AS "titles_key_titles" ON (("keys"."titles__title_key" = "titles_key_titles".title_key)) GROUP BY 1, 2)
SELECT "fk_aggregate_keys"."catalog_overview__year_added" "catalog_overview__year_added", "fk_aggregate_keys"."catalog_overview__genre" "catalog_overview__genre", "q_0"."catalog__count_catalog_titles" "catalog_overview__count_catalog_titles", "q_1"."titles__count_titles" "catalog_overview__count_titles", "q_2"."titles__avg_rating" "catalog_overview__avg_rating"  
FROM (SELECT DISTINCT "catalog_overview__year_added" "catalog_overview__year_added", "catalog_overview__genre" "catalog_overview__genre"  
FROM (SELECT DISTINCT "catalog_overview__year_added" "catalog_overview__year_added", "catalog_overview__genre" "catalog_overview__genre"  
FROM  cte_0  AS "cte_0"
 UNION ALL 
SELECT DISTINCT "catalog_overview__year_added" "catalog_overview__year_added", "catalog_overview__genre" "catalog_overview__genre"  
FROM  cte_1  AS "cte_1"
 UNION ALL 
SELECT DISTINCT "catalog_overview__year_added" "catalog_overview__year_added", "catalog_overview__genre" "catalog_overview__genre"  
FROM  cte_2  AS "cte_2") AS "fk_aggregate_keys_source") AS "fk_aggregate_keys"
LEFT JOIN  cte_0  AS "q_0" ON (("fk_aggregate_keys"."catalog_overview__year_added" = "q_0"."catalog_overview__year_added" OR (("fk_aggregate_keys"."catalog_overview__year_added" IS NULL) AND ("q_0"."catalog_overview__year_added" IS NULL)))) AND (("fk_aggregate_keys"."catalog_overview__genre" = "q_0"."catalog_overview__genre" OR (("fk_aggregate_keys"."catalog_overview__genre" IS NULL) AND ("q_0"."catalog_overview__genre" IS NULL))))
LEFT JOIN  cte_1  AS "q_1" ON (("fk_aggregate_keys"."catalog_overview__year_added" = "q_1"."catalog_overview__year_added" OR (("fk_aggregate_keys"."catalog_overview__year_added" IS NULL) AND ("q_1"."catalog_overview__year_added" IS NULL)))) AND (("fk_aggregate_keys"."catalog_overview__genre" = "q_1"."catalog_overview__genre" OR (("fk_aggregate_keys"."catalog_overview__genre" IS NULL) AND ("q_1"."catalog_overview__genre" IS NULL))))
LEFT JOIN  cte_2  AS "q_2" ON (("fk_aggregate_keys"."catalog_overview__year_added" = "q_2"."catalog_overview__year_added" OR (("fk_aggregate_keys"."catalog_overview__year_added" IS NULL) AND ("q_2"."catalog_overview__year_added" IS NULL)))) AND (("fk_aggregate_keys"."catalog_overview__genre" = "q_2"."catalog_overview__genre" OR (("fk_aggregate_keys"."catalog_overview__genre" IS NULL) AND ("q_2"."catalog_overview__genre" IS NULL)))) ORDER BY  2  ASC,  1  ASC
LIMIT 10000
